cmake_minimum_required(VERSION 3.16)
project(englishvoice LANGUAGES C CXX)
include(flite-sources.cmake)
add_library(flite_core STATIC ${FLITE_SOURCES})
set_target_properties(flite_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(flite_core PRIVATE CST_AUDIO_NONE)
target_include_directories(flite_core PUBLIC flite/include PRIVATE flite/lang/usenglish flite/lang/cmulex)
add_library(englishvoice SHARED englishvoice.cpp)
target_link_libraries(englishvoice PRIVATE flite_core libace_napi.z.so)
